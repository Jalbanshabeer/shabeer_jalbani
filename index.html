OOP Basics: C++ & Java

Class & ObjectBlueprint and its real instance
EncapsulationHide data, expose methods
InheritanceReuse code from a parent class
PolymorphismOne interface, many behaviors
AbstractionShow only what matters
#include <iostream>
using namespace std;

class Animal {
protected: string name;               // encapsulation
public:
  Animal(string n) : name(n) {}
  virtual void speak() { cout << name << " makes a sound\n"; }
};

class Dog : public Animal {   // inheritance
public:
  Dog(string n) : Animal(n) {}
  void speak() override { cout << name << " says Woof!\n"; }
};

int main() {
  Animal* a = new Dog("Tommy");
  a->speak();                          // polymorphism
}
class Animal {
  protected String name;               // encapsulation
  Animal(String name) { this.name = name; }
  void speak() { System.out.println(name + " makes a sound"); }
}

class Dog extends Animal {          // inheritance
  Dog(String name) { super(name); }
  @Override void speak() { System.out.println(name + " says Woof!"); }
}

public class Main {
  public static void main(String[] args) {
    Animal a = new Dog("Tommy");
    a.speak();                          // polymorphism
  }
}
